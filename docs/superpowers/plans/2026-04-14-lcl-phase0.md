```python
from dataclasses import dataclass, field


@dataclass(frozen=True)
class LexiconEntry:
    rank: int
    codepoint: str  # e.g. "U+4E8B"
    symbol: str  # e.g. "事"
    concept_id: str  # e.g. "TASK"
    english_glosses: list[str]  # e.g. ["task", "job"]
    domain: str  # "agentic" | "universal"
    tier: int  # 1, 2, or 3


@dataclass(frozen=True)
class StructuralMarker:
    name: str  # e.g. "MOD"
    codepoint: str  # e.g. "U+25B2"
    symbol: str  # e.g. "▲"


@dataclass(frozen=True)
class Lexicon:
    version: str
    entries: list[LexiconEntry]
    structural_markers: dict[str, StructuralMarker]

    def symbol_to_entry(self) -> dict[str, LexiconEntry]:
        return {e.symbol: e for e in self.entries}

    def concept_to_entry(self) -> dict[str, LexiconEntry]:
        return {e.concept_id: e for e in self.entries}

    def gloss_to_entry(self) -> dict[str, LexiconEntry]:
        result: dict[str, LexiconEntry] = {}
        for entry in self.entries:
            for gloss in entry.english_glosses:
                result[gloss.lower()] = entry
        return result

    def marker_symbols(self) -> set[str]:
        return {m.symbol for m in self.structural_markers.values()}


@dataclass
class ValidationResult:
    is_valid: bool
    total_symbols: int
    known_symbols: int
    unknown_symbols: list[str]
    marker_count: int
    english_fallback_count: int
    warnings: list[str] = field(default_factory=list)


@dataclass
class TransliterateResult:
    english: str
    confidence: float  # 0–1
    unknown_symbols: list[str]
    warnings: list[str] = field(default_factory=list)
```

import json
from pathlib import Path
from lcl.core.types import Lexicon, LexiconEntry, StructuralMarker


def load_lexicon(path: Path) -> Lexicon:
    with open(path) as f:
        data = json.load(f)

    entries = [
        LexiconEntry(
            rank=e["rank"],
            codepoint=e["codepoint"],
            symbol=e["symbol"],
            concept_id=e["concept_id"],
            english_glosses=e["english_glosses"],
            domain=e["domain"],
            tier=e["tier"],
        )
        for e in data["entries"]
    ]

    markers = {
        name: StructuralMarker(
            name=name,
            codepoint=m["codepoint"],
            symbol=m["symbol"],
        )
        for name, m in data["structural_markers"].items()
    }

    return Lexicon(
        version=data["version"],
        entries=entries,
        structural_markers=markers,
    )
```

from lcl.core.types import Lexicon, ValidationResult


def validate(lcl_string: str, lexicon: Lexicon) -> ValidationResult:
    symbol_lookup = lexicon.symbol_to_entry()
    marker_syms = lexicon.marker_symbols()

    eng_start = lexicon.structural_markers["ENG_START"].symbol
    eng_end = lexicon.structural_markers["ENG_END"].symbol

    known = 0
    unknown: list[str] = []
    markers = 0
    eng_fallbacks = 0
    warnings: list[str] = []
    total = 0

    i = 0
    chars = [ch for ch in lcl_string if not ch.isspace()]

    while i < len(chars):
        ch = chars[i]

        if ch == eng_start:
            # Find matching ENG_END
            j = i + 1
            while j < len(chars) and chars[j] != eng_end:
                j += 1
            if j >= len(chars):
                warnings.append("Unmatched ENG_START marker without ENG_END")
            eng_fallbacks += 1
            markers += 2 if j < len(chars) else 1
            i = j + 1
            continue

        if ch in marker_syms and ch != eng_end:
            markers += 1
            i += 1
            continue

        if ch == eng_end:
            warnings.append("Unexpected ENG_END marker without preceding ENG_START")
            markers += 1
            i += 1
            continue

        total += 1
        if ch in symbol_lookup:
            known += 1
        else:
            unknown.append(ch)

        i += 1

    return ValidationResult(
        is_valid=len(unknown) == 0,
        total_symbols=total,
        known_symbols=known,
        unknown_symbols=unknown,
        marker_count=markers,
        english_fallback_count=eng_fallbacks,
        warnings=warnings,
    )
```

import tiktoken


def count_tokens(text: str, encoding: str = "cl100k_base") -> int:
    if not text:
        return 0
    enc = tiktoken.get_encoding(encoding)
    return len(enc.encode(text))


def compression_ratio(english: str, lcl: str, encoding: str = "cl100k_base") -> float:
    eng_tokens = count_tokens(english, encoding)
    lcl_tokens = count_tokens(lcl, encoding)
    if eng_tokens == 0:
        return 0.0
    return 1 - (lcl_tokens / eng_tokens)

    llm_output: str,
    lexicon_symbols: set[str],
    marker_symbols: set[str],

    """Return fraction of output characters that are valid lexicon or marker symbols."""
    chars = [ch for ch in llm_output if not ch.isspace()]
    if not chars:

```python
#!/usr/bin/env python3
"""Phase 0 experiment runner — tests LLM comprehension of LCL encoding.

Usage:
    uv run python scripts/phase0_experiments.py --model llama3.2:3b
    uv run python scripts/phase0_experiments.py --model mistral:7b --host http://localhost:11434

Requires Ollama to be running and the model to be pulled.
"""

import argparse
import json
from pathlib import Path

import ollama

from lcl.core.lexicon import load_lexicon
from lcl.core.tokens import count_tokens

DATA_DIR = Path(__file__).parent.parent / "data"
PROMPTS_DIR = Path(__file__).parent.parent / "prompts"
SCENARIOS_PATH = Path(__file__).parent / "phase0_scenarios.json"


def load_system_prompt() -> str:
    return (PROMPTS_DIR / "phase0-system.md").read_text()


def load_scenarios() -> list[dict]:
    with open(SCENARIOS_PATH) as f:
        return json.load(f)


def run_production_test(
    scenario: dict,
    model: str,
    system_prompt: str,
    client: ollama.Client,
) -> dict:
    user_msg = (
        f"Translate this to LCL (respond ONLY with LCL symbols, no English): "
        f"{scenario['english_instruction']}"
    )
    response = client.chat(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_msg},
        ],
    )
    llm_output = response.message.content.strip()

    return {
        "scenario_id": scenario["id"],
        "input": scenario["english_instruction"],
        "llm_output": llm_output,
        "expected_concepts": scenario["expected_lcl_concepts"],
    }


def run_comprehension_test(
    scenario: dict,
    model: str,
    system_prompt: str,
    client: ollama.Client,
) -> dict:
    user_msg = (
        f"The following is an LCL-encoded instruction. "
        f"Explain in plain English what it means: {scenario['lcl_input']}"
    )
    response = client.chat(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_msg},
        ],
    )
    llm_output = response.message.content.strip()

    return {
        "scenario_id": scenario["id"],
        "lcl_input": scenario["lcl_input"],
        "llm_output": llm_output,
        "expected_meaning": scenario["expected_english_meaning"],
    }


def run_compression_test(scenarios: list[dict]) -> dict:
    total_english_tokens = 0
    total_lcl_tokens = 0

    pairs = []
    for s in scenarios:
        if s["type"] == "comprehension":
            eng_tokens = count_tokens(s["expected_english_meaning"])
            lcl_tokens = count_tokens(s["lcl_input"])
            total_english_tokens += eng_tokens
            total_lcl_tokens += lcl_tokens
            pairs.append({
                "english": s["expected_english_meaning"],
                "lcl": s["lcl_input"],
                "english_tokens": eng_tokens,
                "lcl_tokens": lcl_tokens,
                "ratio": 1 - (lcl_tokens / eng_tokens) if eng_tokens > 0 else 0,
            })

    return {
        "pairs": pairs,
        "total_english_tokens": total_english_tokens,
        "total_lcl_tokens": total_lcl_tokens,
        "overall_compression_ratio": (
            1 - (total_lcl_tokens / total_english_tokens)
            if total_english_tokens > 0
            else 0
        ),
    }


def main():
    parser = argparse.ArgumentParser(description="Phase 0 LCL experiments")
    parser.add_argument("--model", required=True, help="Ollama model name")
    parser.add_argument(
        "--host", default="http://localhost:11434", help="Ollama API host"
    )
    parser.add_argument(
        "--output", default="tests/phase0/raw_results.json", help="Output file"
    )
    args = parser.parse_args()

    client = ollama.Client(host=args.host)
    system_prompt = load_system_prompt()
    scenarios = load_scenarios()
    lexicon = load_lexicon(DATA_DIR / "phase0-lexicon.json")

    symbol_set = set(lexicon.symbol_to_entry().keys())
    marker_set = lexicon.marker_symbols()

    print(f"Running Phase 0 experiments with model: {args.model}")
    print(f"Lexicon: {len(lexicon.entries)} entries")
    print()

    results = {
        "model": args.model,
        "lexicon_version": lexicon.version,
        "production_tests": [],
        "comprehension_tests": [],
        "compression": None,
    }

    # Production tests
    prod_scenarios = [s for s in scenarios if s["type"] == "production"]
    print(f"--- Production Tests ({len(prod_scenarios)} scenarios) ---")
    for scenario in prod_scenarios:
        print(f"  {scenario['id']}: {scenario['english_instruction']}")
        result = run_production_test(scenario, args.model, system_prompt, client)
        print(f"    LLM output: {result['llm_output']}")

        # Score: what fraction of output chars are valid symbols?
        output_chars = [ch for ch in result["llm_output"] if not ch.isspace()]
        valid = sum(1 for ch in output_chars if ch in symbol_set or ch in marker_set)
        score = valid / len(output_chars) if output_chars else 0
        result["valid_symbol_ratio"] = score
        print(f"    Valid symbol ratio: {score:.2%}")
        results["production_tests"].append(result)

    print()

    # Comprehension tests
    comp_scenarios = [s for s in scenarios if s["type"] == "comprehension"]
    print(f"--- Comprehension Tests ({len(comp_scenarios)} scenarios) ---")
    for scenario in comp_scenarios:
        print(f"  {scenario['id']}: {scenario['lcl_input']}")
        result = run_comprehension_test(scenario, args.model, system_prompt, client)
        print(f"    LLM interpretation: {result['llm_output'][:100]}")
        results["comprehension_tests"].append(result)

    print()

    # Compression tests
    print("--- Compression Tests ---")
    compression = run_compression_test(scenarios)
    results["compression"] = compression
    for pair in compression["pairs"]:
        print(
            f"  '{pair['english'][:40]}...' → '{pair['lcl']}': "
            f"{pair['english_tokens']}→{pair['lcl_tokens']} tokens "
            f"({pair['ratio']:.0%} reduction)"
        )
    print(f"  Overall compression: {compression['overall_compression_ratio']:.0%}")

    # Measure system prompt token cost
    prompt_tokens = count_tokens(system_prompt)
    results["system_prompt_tokens"] = prompt_tokens
    print(f"\n  System prompt cost: {prompt_tokens} tokens")

    # Save results
    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(results, f, indent=2, ensure_ascii=False)
    print(f"\nResults saved to {output_path}")


if __name__ == "__main__":
    main()
```
